// 1. Estrutura if-then
// if (A>=B) { A = A + 1 }
    LD A         
    SUB B        
    BLT FIM_1  
    LD A        
    ADDI 1       
    STO A        
FIM_1:
    HLT


// 2. Estrutura if-then-else
// if (A>B) { A = A+1; } else { B = B+1; }
    LD A        
    SUB B
    BLE ELSE_2    
    LD A
    ADDI 1       
    STO A
    JMP FIM_2   
ELSE_2:
    LD B
    ADDI 1       
    STO B
FIM_2:
    HLT


// 3. Estrutura while
// i = 0;
// while (i<10) { A += 2; i++; }
    LDI 0         
    STO I        
WHILE_3:
    LD I
    SUBI 10      
    BGE FIM_3
    LD A
    ADDI 2
    STO A        
    LD I
    ADDI 1
    STO I        
    JMP WHILE_3
FIM_3:
    HLT


// 4. Estrutura do-while
// i = 0; 
// do { A += 2; i++; } while (i<10)
    LDI 0         
    STO I     
DO_4:
    LD A
    ADDI 2
    STO A        
    LD I
    ADDI 1
    STO I        
    LD I
    SUBI 10     
    BLT DO_4 
    HLT


// 5. Estrutura if / else if / else if
// if (a == b) a = a + 1;
// else if (a > b) a = a + 2;
// else if (a < b) a = a + 3;
    LD A
    SUB B
    BEQ IGUAL_5
    BGT MAIOR_5
    LD A
    ADDI 3
    STO A
    JMP FIM_5
IGUAL_5:
    LD A
    ADDI 1
    STO A
    JMP FIM_5
MAIOR_5:
    LD A
    ADDI 2
    STO A
    JMP FIM_5
FIM_5:
    HLT
